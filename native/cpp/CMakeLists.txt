cmake_minimum_required(VERSION 3.22)
project(WinArchiveFSPro LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)
find_package(LibArchive REQUIRED)
function(archivefs_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
if(WIN32)
  set(ARCHIVEFS_LIBRARY_TYPE SHARED)
else()
  set(ARCHIVEFS_LIBRARY_TYPE STATIC)
endif()
add_library(archivefs_core ${ARCHIVEFS_LIBRARY_TYPE} random_access_source.cpp archive_reader.cpp decoder_backend.cpp archive_tree.cpp process.cpp image_creation.cpp projection.cpp)
if(WIN32)
  set_target_properties(archivefs_core PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()
target_compile_features(archivefs_core PUBLIC cxx_std_23)
target_include_directories(archivefs_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(archivefs_core PUBLIC Threads::Threads ${CMAKE_DL_LIBS} LibArchive::LibArchive)
archivefs_warnings(archivefs_core)
option(ARCHIVEFS_WITH_DWARFS "Use the upstream DwarFS C++ reader" OFF)
if(ARCHIVEFS_WITH_DWARFS)
  find_package(dwarfs 0.15.10 EXACT CONFIG REQUIRED)
  target_sources(archivefs_core PRIVATE dwarfs_backend.cpp)
  target_compile_definitions(archivefs_core PRIVATE ARCHIVEFS_WITH_DWARFS=1)
  target_link_libraries(archivefs_core PRIVATE dwarfs::dwarfs_reader dwarfs::dwarfs_common)
endif()

if(WIN32)
  target_compile_definitions(archivefs_core PRIVATE _WIN32_WINNT=0x0A00)
  target_link_libraries(archivefs_core PRIVATE bcrypt)
else()
  find_package(OpenSSL REQUIRED)
  target_link_libraries(archivefs_core PRIVATE OpenSSL::Crypto)
endif()
add_executable(archivefs-read cli.cpp)
if(WIN32 AND MSVC)
  target_link_options(archivefs-read PRIVATE "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/gui/app.manifest" /MANIFEST:EMBED)
endif()
target_link_libraries(archivefs-read PRIVATE archivefs_core)
archivefs_warnings(archivefs-read)
if(MINGW)
  target_link_options(archivefs-read PRIVATE -municode)
endif()
if(BUILD_TESTING)
  foreach(test concurrent_cache archive_reader decoder_backend)
    add_executable(${test}_tests tests/${test}.cpp)
    target_link_libraries(${test}_tests PRIVATE archivefs_core)
    archivefs_warnings(${test}_tests)
  endforeach()
  foreach(format winsqfs-native winsqfs-dwarfs winsqfs-erofs)
    add_library(mock_${format} SHARED tests/mock_decoder.cpp)
    set_target_properties(mock_${format} PROPERTIES OUTPUT_NAME ${format}
      RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/mock"
      LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/mock")
    target_compile_features(mock_${format} PRIVATE cxx_std_23)
    archivefs_warnings(mock_${format})
  endforeach()
  add_dependencies(decoder_backend_tests mock_winsqfs-native mock_winsqfs-dwarfs mock_winsqfs-erofs)
  add_test(NAME concurrent_cache COMMAND concurrent_cache_tests)
  add_test(NAME archive_reader COMMAND archive_reader_tests)
  add_test(NAME decoder_backend COMMAND decoder_backend_tests "$<TARGET_FILE_DIR:mock_winsqfs-native>")
  set_tests_properties(concurrent_cache archive_reader decoder_backend PROPERTIES TIMEOUT 30)
endif()
if(BUILD_TESTING)
